import numpy as np
import matplotlib.pyplot as plt
from matplotlib.patches import Circle
from dataclasses import dataclass
from toolbox.image_helper import Imgtool
from toolbox.poly_fitting_helper import Poly_Fitting

@dataclass
class Config:
    width: int = 640
    height: int = 480
    center_x: float = 320.0
    center_y: float = 240.0
    radius: float = 150.0
    dx: float = 0.0
    dy: float = 0.0
    supersample: int = 4
    blur_size: int = 7
    blur_sigma: float = 0.8
    canny_low: float = 50
    canny_high: float = 150

def renderer(cfg, target=False):
    ss = cfg.supersample
    x = (np.arange(cfg.width * ss) + 0.5) / ss - 0.5
    y = (np.arange(cfg.height * ss) + 0.5) / ss - 0.5
    xx, yy = np.meshgrid(x, y)
    cx = cfg.center_x + (cfg.dx if target else 0.0)
    cy = cfg.center_y + (cfg.dy if target else 0.0)
    u = (xx - cx)**2 + (yy - cy)**2
    image = np.where(u <= cfg.radius**2, 255.0, 0.0)
    image = image.reshape(cfg.height, ss, cfg.width, ss).mean(axis=(1, 3))
    tool = Imgtool(np.rint(image)).cvt_cv2_img(bound='clip', mode='gray')
    if cfg.blur_sigma > 0:
        tool.smooth(size=cfg.blur_size, sigma=cfg.blur_sigma)
    return tool.get()

def fit_circle_contour_xld(pts):
    pts = np.asarray(pts, dtype=float)
    x = pts[:, 0]
    y = pts[:, 1]
    y_target = x**2 + y**2
    custom_order = [(1, 0), (0, 1), (0, 0)]
    model = Poly_Fitting(basis="泰勒")
    model.learn(pts, y_target, order=custom_order, times=1, rescale=False)
    w0, w1, w2 = model.w
    cx = w0 / 2.0
    cy = w1 / 2.0
    radius = np.sqrt(w2 + cx**2 + cy**2)
    return cx, cy, radius, model

def detect_circle(cfg, image):
    edges = Imgtool(image).canny(low=cfg.canny_low, high=cfg.canny_high)
    ys, xs = np.nonzero(edges)
    pts = np.column_stack((xs, ys)).astype(float)
    cx, cy, radius, model = fit_circle_contour_xld(pts)
    return cx, cy, radius, edges, pts

def draw_result(cfg, image, target=False, est_cx=None, est_cy=None, est_radius=None, pts=None):
    fig, axes = plt.subplots(1, 2, figsize=(14, 7))
    ax_overview, ax_zoom = axes
    cx = cfg.center_x + (cfg.dx if target else 0.0)
    cy = cfg.center_y + (cfg.dy if target else 0.0)
    for ax in axes:
        ax.imshow(image, cmap='gray', vmin=0, vmax=255, interpolation='none')
        analytic_circle = Circle((cx, cy), cfg.radius, edgecolor='green', facecolor='none', linewidth=1.5, linestyle='-')
        ax.add_patch(analytic_circle)
        if est_cx is not None and est_cy is not None and est_radius is not None:
            est_circle = Circle((est_cx, est_cy), est_radius, edgecolor='red', facecolor='none', linewidth=1.2, linestyle='--')
            ax.add_patch(est_circle)
        if pts is not None:
            ax.scatter(pts[:, 0], pts[:, 1], s=1, color='blue')
        ax.set_xlabel("x (px)")
        ax.set_ylabel("y (px)")
    ax_overview.set_title("Overview")
    zoom_radius = 20
    ax_zoom.set_xlim(cx + cfg.radius - zoom_radius, cx + cfg.radius + zoom_radius)
    ax_zoom.set_ylim(cy + zoom_radius, cy - zoom_radius)
    ax_zoom.set_title("Zoomed Edge (Right)")
    plt.tight_layout()
    plt.show()

def run_pair(cfg, show=True):
    image0 = renderer(cfg, target=False)
    image1 = renderer(cfg, target=True)
    cx0, cy0, radius0, edges0, pts0 = detect_circle(cfg, image0)
    cx1, cy1, radius1, edges1, pts1 = detect_circle(cfg, image1)
    est_dx = cx1 - cx0
    est_dy = cy1 - cy0
    print(f"True dx={cfg.dx:.6f}, dy={cfg.dy:.6f}")
    print(f"Est  dx={est_dx:.6f}, dy={est_dy:.6f}")
    print(f"Err  dx={est_dx - cfg.dx:.6f}, dy={est_dy - cfg.dy:.6f}")
    if show:
        draw_result(cfg, image0, target=False, est_cx=cx0, est_cy=cy0, est_radius=radius0, pts=pts0)
        draw_result(cfg, image1, target=True, est_cx=cx1, est_cy=cy1, est_radius=radius1, pts=pts1)
    return cx0, cy0, cx1, cy1, est_dx, est_dy

def main():
    cfg = Config()
    cfg.dx = 15.3
    cfg.dy = -8.7
    run_pair(cfg)
if __name__ == '__main__':
    main()
